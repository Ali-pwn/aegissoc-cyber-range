// High-speed C++20 Firewall & DPI Filter
#include <unordered_set>
#include <string>

class PacketFilter {
    std::unordered_set<std::string> blacklist;
public:
    void blockIP(const std::string& ip) {
        blacklist.insert(ip);
    }
    bool inspectPacket(const std::string& srcIP, 
                       const std::string& payload) {
        if (blacklist.count(srcIP)) return false; // DROP
        if (payload.find("' OR 1=1") != std::string::npos) {
            return false; // SQLi Signature Blocked
        }
        return true; // ALLOW
    }
};
